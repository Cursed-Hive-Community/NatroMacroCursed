# Things no well-behaved macro would ever send.
#
# None of this should take the coordinator down, and none of it should be
# obeyed. Watch the fleet log for what it made of each one.

# the seats go in before the coordinator reads the roster,
# rather than relying on it to notice a reload
seat 1 main bench main
coordinator start
wait 2500
connect 1
wait 1500

say --- a verb that does not exist ---
send 1 WOBBLE x=1
wait 500

say --- a frame with no fields ---
send 1 FIELD
wait 500

say --- a field with no value, and a value with no field ---
send 1 FIELD name=
wait 500
send 1 FIELD =pepper
wait 500

say --- somebody else's configuration, at a version from the past ---
send 1 SETROW v=1 by=9 n=1 row=1 name=hijack role=main user= machine=nowhere
wait 2000
say the fleet log should say it ignored it - ours is higher
reject 1 BYE

say --- percent, space and equals inside one value ---
field 1 blue flower
wait 1000
expect 1 ROSTER row=1
